func replace<Element>(mut place: Element, consume replacement: Element) -> Element

pub func replace<Element>(mut place: Element, consume replacement: Element) -> Element {
	let previous: Element = #_ir(place) { %? = take Element $0 }
	place = _take_owned(value: consume replacement)
	previous
}
