pub func render_symbol_pages(
	package: String,
	module: String,
	module_page: String,
	collection: &DocumentableCollection,
	modules: &[ModuleLink],
	symbols: &[SymbolLink]
) -> [RenderedPage] {
	let pages: [RenderedPage] = []
	let views = entry_views(
		collection: collection,
		module: module,
		page: module_page,
		symbols: symbols,
		documentation_heading_offset: 1
	)
	let catalog = catalog_navigation(
		package: package.clone(),
		modules: modules,
		current: module.clone()
	).into_string()
	let wip = wip_banner(
		description: collection.module_wip,
		module: module,
		page: module_page,
		symbols: symbols
	)
	let view_index = 0
	for entry in collection.items {
		if is_top_level(entry: entry) {
			let view = views[view_index]
			let file = symbol_page(
				module_page: module_page,
				signature: view.signature.clone()
			)
			pages.push(
				RenderedPage(
					file: file,
					contents: render_symbol_view(
						package: package.clone(),
						module: module.clone(),
						name: declaration_name(entry: entry),
						view: view,
						catalog_html: catalog,
						wip_html: wip
					)
				)
			)
			view_index = view_index + 1
		}
	}
	pages
}
