pub func symbol_file(module_page: &String, collection: &DocumentableCollection, entry: &DocumentableEntry) -> String? {
	if is_top_level(entry: entry) == false { return .none }
	.some(symbol_page(module_page: module_page, signature: signature_of(collection: collection, entry: entry)))
}
