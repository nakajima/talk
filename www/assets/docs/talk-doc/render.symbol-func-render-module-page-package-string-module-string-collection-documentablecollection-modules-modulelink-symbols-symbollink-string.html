func render_module_page(package: String, module: String, collection: &DocumentableCollection, modules: &[ModuleLink], symbols: &[SymbolLink]) -> String

Renders a module overview followed by categorized API summaries.

Each summary contains the declaration name, linked signature, and first documentation paragraph. Full documentation, members, and source live on the standalone page produced by render_symbol_pages. Categories use a stable semantic order while summaries within each category retain source order.

pub func render_module_page(package: String, module: String, collection: &DocumentableCollection, modules: &[ModuleLink], symbols: &[SymbolLink]) -> String {
	let page = ""
	for link in modules {
		if link.name == module { page = link.file }
	}
	let summaries = summary_views(collection: collection, module: module, page: page, symbols: symbols)
	let wip = PreEscaped(value: wip_banner(description: collection.module_wip, module: module, page: page, symbols: symbols))
	let module_documentation = PreEscaped(value: render_documentation(documentation: collection.module_documentation, heading_offset: 1, module: module, owner: "", current_page: page, symbols: symbols))
	let entries = PreEscaped(value: render_summaries(summaries: summaries))
	let catalog = PreEscaped(value: catalog_navigation(package: package.clone(), modules: modules, current: module.clone()).into_string())
	let page_index = PreEscaped(value: summary_navigation(summaries: summaries).into_string())
	let header = PreEscaped(value: module_header(package: package.clone(), module: module.clone()))
	let shell = PreEscaped(value: module_shell(catalog: catalog, wip: wip, module_documentation: module_documentation, entries: entries, page_index: page_index))
	let content = @html { main class="site-page docs-proposal docs-design-1" { (header) (shell) } }
	page_html(title: module + " - " + package, content: content.into_string())
}
