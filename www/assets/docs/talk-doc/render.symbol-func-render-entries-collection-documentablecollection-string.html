pub func render_entries(collection: &DocumentableCollection) -> String {
	let symbols = symbol_links(module: "", page: "", collection: collection)
	let entries = entry_views(
		collection: collection,
		module: "",
		page: "",
		symbols: symbols,
		documentation_heading_offset: 3
	)
	render_entry_views(entries: entries)
}
