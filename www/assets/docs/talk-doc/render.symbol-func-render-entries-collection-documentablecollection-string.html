func render_entries(collection: &DocumentableCollection) -> String

Renders declaration sections without page chrome or module navigation.

Top-level entries are grouped by declaration kind and retain source order within each group; nominal members nest under their owner. Attached comments are parsed as Markdown and inserted as trusted rendered markup.

pub func render_entries(collection: &DocumentableCollection) -> String {
	let symbols = symbol_links(module: "", page: "", collection: collection)
	let entries = entry_views(collection: collection, module: "", page: "", symbols: symbols, documentation_heading_offset: 3)
	render_entry_views(entries: entries)
}
