func parse(markdown: &String, extensions: &[Extension]) -> Document

Parses a complete Markdown string with an explicit extension set.

Parsing is deterministic and dependency-free. Unsupported or disabled syntax remains ordinary text rather than producing diagnostics. The result drops source trivia and is intended for rendering, not source-to-source editing.

pub func parse(markdown: &String, extensions: &[Extension]) -> Document {
	let lines: [String] = []
	for line in markdown.lines() {
		lines.push(line.to_string())
	}
	let config = config_for(extensions: extensions)
	Document(blocks: parse_blocks(lines: lines, start: 0, end: lines.count, config: config))
}
