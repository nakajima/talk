func append(mut buf: [Byte], text: String) -> Void

Appends the UTF-8 bytes of text to a dump buffer.

View Source
pub func append(mut buf: [Byte], text: String) -> Void {
	let bytes = text.utf8()
	let i = 0
	loop i < bytes.count() {
		buf.push(bytes.at(index: i))
		i = i + 1
	}
}
