func indent(mut buf: [Byte], depth: Int) -> Void

Appends indentation for the requested nesting depth.

View Source
pub func indent(mut buf: [Byte], depth: Int) -> Void {
	let d = 0
	loop d < depth {
		append(buf: buf, text: "  ")
		d = d + 1
	}
}
