pub func parse_file_docs_source(source: String) -> ParseOutcome {
	let outcome = parse_file_source(source: source)
	let docs = collect_doc_comments(source: source, outcome: outcome)
	ParseOutcome(items: outcome.items, failure: outcome.failure, diags: outcome.diags, comments: outcome.comments, metas: outcome.metas, docs: docs.entries)
}
