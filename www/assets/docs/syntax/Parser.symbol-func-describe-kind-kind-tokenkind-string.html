pub func describe_kind(kind: TokenKind) -> String {
	if kind == .eof { return "end of input" }
	if kind == .newline { return "a newline" }
	if kind == .identifier { return "an identifier" }
	if kind == .int_number { return "an integer literal" }
	if kind == .float_number { return "a float literal" }
	if kind == .string_literal { return "a string literal" }
	if kind == .character_literal { return "a character literal" }
	"`" + kind_spelling(kind: kind) + "`"
}
